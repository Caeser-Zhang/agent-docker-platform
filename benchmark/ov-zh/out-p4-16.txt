#!/bin/bash
# Probe 4: MCP hiding via preset mcps exclusion + global top-level deny.
#  (a) modify plugin config preset: orchestrator mcps ["*"] -> ["*","!fastk"]
#  (b) PATCH global config permission fastk_* -> deny (real change, triggers reload)
#  (c) GET /agent: orchestrator fastk_* should be deny (tail rule from plugin loop);
#      native "build" agent fastk_* should be deny (from global permission).
#  (d) restore both.
set -u
C="${1:-agent-f4f5d8ce-f63d-475b-a083-5a0113f66a9b}"
PORT=4096
TMP=$(mktemp)
OMO="/data/config/opencode/oh-my-opencode-slim.json"

get() {
  docker exec "$C" bash -c "curl -s -u \"opencode:\$OPENCODE_SERVER_PASSWORD\" 'http://127.0.0.1:$PORT$1'" > "$TMP"
}
patch() {
  docker exec "$C" bash -c "curl -s -o /dev/null -w '%{http_code}' -u \"opencode:\$OPENCODE_SERVER_PASSWORD\" -X PATCH -H 'content-type: application/json' -d '$1' 'http://127.0.0.1:$PORT/global/config'"
}
py() {  # py SCRIPT  — run python inside container
  docker exec "$C" python3 -c "$1"
}
perm_of() {  # perm_of AGENTNAME -> print mcp-ish rules of agent
  python3 - "$TMP" "$1" <<'PYEOF'
import json, sys
d = json.load(open(sys.argv[1]))
name = sys.argv[2]
a = next((x for x in d if isinstance(x, dict) and x.get("name") == name), None)
if not a:
    print(f"  ({name}: NOT FOUND)"); sys.exit()
for i, r in enumerate(a.get("permission") or []):
    p = r.get("permission") or ""
    if p in ("*", "fastk_*", "demo-mcp_*", "web_search_*", "context7_*", "gh_grep_*", "skill", "skill:*"):
        print(f"  {name}[{i:2d}] {p:14s} -> {r.get('action')}")
PYEOF
}

py "import json; c=json.load(open('$OMO'));
print(json.dumps({k:{'mcps':v.get('mcps'),'skills':v.get('skills')} for k,v in (c.get('presets',{}).get('container') or {}).items()}, indent=1));
print('disabled_skills:', c.get('disabled_skills'))"

echo

import json
